Tích Hợp Dễ Dàng Trong 5 Dòng Code

Hỗ trợ mọi ngôn ngữ: Python, Node.js, PHP/Laravel, Golang hoặc cURL.

# Sinh mã VietQR chuyển khoản tự động
import requests

payload = {
    "bank_code": "TPB",
    "account_number": "20058999999",
    "account_name": "LE NGUYEN HUY HOANG",
    "amount": 50000,
    "order_id": "DH8921",
    "webhook_url": "https://your-domain.com/api/payment-callback"
}

res = requests.post("http://localhost:8000/api/v1/vietqr/generate", json=payload)
qr_data = res.json()
print("QR Image URL:", qr_data["qr_image_url"])
print("EMVCo Raw:", qr_data["emvco_payload"])
